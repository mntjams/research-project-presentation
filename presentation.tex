\section{Introduction}
\begin{frame}
    \frametitle{ELMM}
    \begin{itemize}
        \item Extended Large Eddy Microscale Model (ELMM) is a model for atmospheric flows
        \item Written in Fortran
        \item Parallelized using OpenMP, MPI, or hybrid MPI + OpenMP\pause
        \note {
            - Porting this code to GPUs could greatly benefit its performance
        }
        \item In each time step a \alert{Poisson equation} must be solved\pause
    \end{itemize}
    \begin{figure}
        \centering
        \includegraphics[width=0.5\textwidth]{ELMM-Simulation.png}
    \end{figure}
\end{frame}

\begin{frame}
    \frametitle{PoisFFT - Poisson solver}
    \begin{itemize}
        \item Solves the Poisson equation using FFT-based mathods
        \item Parallelized the same way as ELMM
        \item Best starting point for ELMM performance improvement
        \note{
            - It is a self-contained library
            - Takes up noticable portion of the run-time
        }
        \note {
            - While measuring the performance, we encountered several hybrid configurations where PoisFFT straight up segfaulted
            - See next slide for configuration definition
            - With no existing documentation and PFFTs (PoisFFT multi-node FFT library) hybrid option being in development for the past 10 years, we gave up on that
        }
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Outside view of PoisFFT}
    \begin{itemize}
        \item User defines a domain:\pause
            \begin{itemize}
                \item Dimensionality
                \item Number of grid cells
                \item Grid lengths
                \item Boundary conditions\pause
            \end{itemize}
        \item Then (repeatedly) solves the equation by supplying $b$
        \note{
            - $b$ is just an array of size (ns_x, ns_y, ns_z)
            - Both $b$ and $\phi$ can be surrounded by ghost cells
        }
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Inside view of PoisFFT}
    PoisFFT has two phases:\pause
    \begin{enumerate}
        \item \alert{initialization}, and\pause
            \begin{itemize}
                \item Precompute the normalization factor and eigenvalues
                \item Prepare FFT plan (or plans)
                \item Allocate workspaces\pause
            \end{itemize}
        \item (repeated) \alert{execution}\pause
            \begin{itemize}
                \item Apply forward FFT
                \item Divide every element by a sum of pre-computed eigenvalues
                \item Apply backward FFT
                \item Normalize the result
                \note {
                    - The steps differ a bit depending on the configuration
                }
            \end{itemize}
    \end{enumerate}
\end{frame}

\begin{frame}
    \frametitle{PoisFFT - Problems}
    \begin{itemize}
        \item PoisFFT has several problems:\pause
            \begin{itemize}
                \item Tens of configurations\pause
                \item No documentation\pause
                \item Limited usage guide\pause
                \item Poor architecture design\pause
                \item Sparsely commeted code\pause
                \item Bad code hygiene
            \end{itemize}
        \note {
            - D is a solver
            - D1D is a 1D solver, but we never use it to solve anything, it is used to just hold data
            - tmp1, tmp2, ctmp1, ctmp2
        }
        \note {
            - It is entirely possible to create a solver without an error but the solver does nothing on each Execute
            - They are interleaved with ifdefs to form if-else blocks based on compile time definitions
        }
    \end{itemize}
\end{frame}

\section{poisson-solver}
\begin{frame}
    \frametitle{poisson-solver — a modern PoisFFT port}
    \begin{alertblock}{Our goal}
        Create a PoisFFT GPU port with better performance and better software architecture\pause
    \end{alertblock}
    How to achieve that?\pause
    \begin{enumerate}
        \item Move the hot path to a GPU\pause
        \item Replace FFT libraries with GPU-based alternatives\pause
        \item Improve separation of concerns\pause
        \item Write documentation
    \end{enumerate}
\end{frame}

\begin{frame}
    \frametitle{The architecture of poisson-solver}
    \begin{itemize}
        \item poisson-solver makes a clear distinction between:\pause
        \begin{itemize}
            \item Configuration-based decisions
            \item Implications of those decisions
            \item Compile-time setup\pause
        \end{itemize}
    \end{itemize}
    % TODO - Make this into a schema that summarizes what is inside each block and how the block relate to each other (you can also invent new blocks that you deem necessary)
    % This TODO also applies to the next frame/slide
    \begin{block}{Solver plan}
        A configuration-dependant plan that the rest of the solver follows.
        Includes: the solve method, strategies for eigenvalue computation, reshape types, etc.
        \note {
            - Reshapes are used in MPI code to get one axis into a local memory
            - Reshape types are e.g. (to pencil, to slab)
        }
    \end{block}\pause
    \begin{block}{Solver core}
        Owns objects, buffers and precomputed values specified by the plan.
        E.g., precomputed eigenvalues, FFT plan, the reshaper or, device workspaces.
        \note {
            - FFT libraries usually work by first defining a plan which optimizes for the specified FFT usage
            - Then, repeatedly executing such plan
        }
    \end{block}
\end{frame}

\begin{frame}
    \frametitle{The architecture of poisson-solver}
    \begin{block}{FFT policies}
        Policies describing how to perform the FFTs depending on the build type.
        One is purely local, the other also works in distributed environments.
        \note {
            - Different libraries are used for different build types
        }
    \end{block}\pause
    \begin{block}{Reshaper}
        Certain steps of the execution pipeline require data locality.
        A reshaper object performs transitions between decompositions.
        \note {
            - The decompositions and transitions between them are precomputed during the solver core construction
        }
    \end{block}
\end{frame}

\section{Performance}
\begin{frame}
    \frametitle{Measurements}
    \begin{itemize}
        \item PoisFFT \pause vs poisson-solver (host-device copies) \pause vs poisson-solver (device-device copies)\pause
    \end{itemize}
    \begin{block}{Local solver}
        \begin{itemize}
            \item poisson-solver (device-device copies) always wins!
            \begin{itemize}
                \item 2.4x to 10.4x speedups
                \item Maximum speedup 113x\pause
            \end{itemize}
            \item poisson-solver (host-device copies) wins in $\sim 60\%$ of cases
            \begin{itemize}
                \item 0.75x to 1.80x speedups\pause
            \end{itemize}
        \end{itemize}
    \end{block}
    \begin{exampleblock}{Distributed solver}
        \begin{itemize}
            \item poisson-solver (device-device copies) always wins!
            \begin{itemize}
                \item 1.39x to 2.6x speedups\pause
            \end{itemize}
            \item poisson-solver (host-device copies) always wins!
            \begin{itemize}
                \item 1.31x to 2.36x speedups
            \end{itemize}
        \end{itemize}
    \end{exampleblock}
\end{frame}

\begin{frame}
    \frametitle{Computational fluid dynamics}
    \begin{block}{Navier-Stokes equations}
        The Navier-Stokes equations are
        \[
            \rho (\frac{\delta \mathbf{u}}{\delta t} + \mathbf{u} \cdot \nabla \mathbf{u}) = -\nabla p + \mu\nabla^2 \mathbf{u} + F
        \]
        \begin{itemize}
            \item fluid density $\rho$
            \item velocity vector field $\mathbf{u}$
            \item time $t$
            \item pressure field $p$
            \item dynamic viscosity $\mu$
            \item external forces $F$
        \end{itemize}
    \end{block}
    \begin{block}{Poisson equation}
        The Poisson equation is
        \[
            \Delta \phi = b
        \]
        \begin{itemize}
            \item pressure change $\phi$
            \item net outflow of predicted wind $b$
        \end{itemize}
    \end{block}
\end{frame}
